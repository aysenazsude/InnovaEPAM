## 3. New Module — `src/lib/attachments/mimeDetector.ts`

Pure function, no external dependencies. Returns the detected MIME type string or `null` if unrecognized.

```typescript
// src/lib/attachments/mimeDetector.ts — NEW

/** Magic-byte signatures for all accepted file types (Phase 3). */
const SIGNATURES: ReadonlyArray<{
  mime: string;
  offset: number;
  bytes: readonly number[];
}> = [
  { mime: 'application/pdf',   offset: 0, bytes: [0x25, 0x50, 0x44, 0x46] },          // %PDF
  { mime: 'application/zip',   offset: 0, bytes: [0x50, 0x4B, 0x03, 0x04] },          // PK (DOCX/PPTX)
  { mime: 'application/msword',offset: 0, bytes: [0xD0,0xCF,0x11,0xE0,0xA1,0xB1,0x1A,0xE1] }, // DOC
  { mime: 'image/png',         offset: 0, bytes: [0x89,0x50,0x4E,0x47,0x0D,0x0A,0x1A,0x0A] },
  { mime: 'image/jpeg',        offset: 0, bytes: [0xFF, 0xD8, 0xFF] },
  { mime: 'video/mp4-or-mov',  offset: 4, bytes: [0x66, 0x74, 0x79, 0x70] },          // ftyp box
];

const ZIP_DOCX_MIME = 'application/vnd.openxmlformats-officedocument.wordprocessingml.document';
const ZIP_PPTX_MIME = 'application/vnd.openxmlformats-officedocument.presentationml.presentation';

/**
 * Detects the MIME type of a file buffer by inspecting its magic bytes.
 * Returns the MIME string or null if unrecognized.
 */
export function detectMimeType(buffer: Buffer, fileExtension: string): string | null {
  for (const sig of SIGNATURES) {
    const slice = buffer.subarray(sig.offset, sig.offset + sig.bytes.length);
    if (sig.bytes.every((b, i) => slice[i] === b)) {
      if (sig.mime === 'application/zip') {
        // Disambiguate DOCX vs PPTX by extension (both are valid)
        const ext = fileExtension.toLowerCase();
        if (ext === '.pptx') return ZIP_PPTX_MIME;
        return ZIP_DOCX_MIME; // default ZIP → DOCX
      }
      if (sig.mime === 'video/mp4-or-mov') {
        // Both MP4 and MOV are accepted — return canonical MIME by extension
        return fileExtension.toLowerCase() === '.mov' ? 'video/quicktime' : 'video/mp4';
      }
      return sig.mime;
    }
  }
  return null;
}
```

## 4. Updated Module — `src/lib/attachments/attachmentValidator.ts`

### 4.1 New `validateAttachments` (multi-file) signature

```typescript
export interface AttachmentFileInput {
  name: string;
  mimeType: string;    // from detectMimeType()
  sizeBytes: number;
}

export interface MultiAttachmentValidationResult {
  valid: boolean;
  /** Per-file errors keyed by file name */
  fileErrors: Record<string, string>;
  /** Submission-level errors (total size, count) */
  formErrors: Record<string, string>;
}

export function validateAttachments(
  files: AttachmentFileInput[]
): MultiAttachmentValidationResult
```

### 4.2 Validation Rules Encoded

| Check | Rule | Error target |
|-------|------|-------------|
| Count | `files.length ≤ MAX_ATTACHMENTS_PER_IDEA` | `formErrors['files']` |
| Per-file type | `ALLOWED_MIME_TYPES.includes(mimeType)` | `fileErrors[name]` |
| Per-file size | `sizeBytes ≤ FILE_SIZE_LIMIT` (10 MB) | `fileErrors[name]` |
| Duplicate names | No two files share the same `name` | `fileErrors[name]` (second) |
| Total size | `sum(sizeBytes) ≤ MAX_TOTAL_ATTACHMENT_SIZE` (30 MB) | `formErrors['totalSize']` |

### 4.3 Legacy `validateAttachment` (single-file)

Kept unchanged for the Phase 1 `POST /api/attachments` route (backwards compatibility).

## 5. New Module — `src/lib/attachments/mimeToIcon.ts`

Maps MIME type to a display category used by `FileUpload` and admin detail for icon/preview selection.

```typescript
export type PreviewKind = 'image' | 'video' | 'document';

export function getPreviewKind(mimeType: string): PreviewKind {
  if (mimeType === 'image/png' || mimeType === 'image/jpeg') return 'image';
  if (mimeType === 'video/mp4' || mimeType === 'video/quicktime') return 'video';
  return 'document';
}
```

```typescript
  file: File;

  kind: PreviewKind;
  error: string | null;
}

// Component state

```

- `previewUrl` lifetime: created on add, revoked in `useEffect` cleanup and on individual remove.
- Max length: `MAX_ATTACHMENTS_PER_IDEA` (3) — add control disabled when `staged.length >= 3`.
- Submitted to server: one hidden `<input type="file" name="file">` per staged file inside the form (using `DataTransfer` to programmatically populate inputs) **or** the files are passed via `FormData.append('file', file)` in a `fetch` call — see R2 in research.md.
